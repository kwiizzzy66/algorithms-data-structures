#include <iostream>

#пользователь вводит значение n
std::cout << "Введите n: ";
std::cin >> n;

#Проверяем, что n не меньше 1
if (n < 1) {
    #Если n меньше 1, выводит ошибку
    std::cout << "n должно быть >= 1\n"
    #Завершаем программу с кодом ошибки
    return 1;
}

#Используем "unsigned long long" для работы с большими числами
unsigned long long F = 1;  #Здесь будет храниться произведение факториалов
unsigned long long factorial = 1;  #Текущее значение факториала

#Цикл для вычисления факториала от 1 до n
for (int i = 1; i <= n; i++) {
    #Вычисляем факториал текущего числа i
    factorial *= i;
    #Умножаем F на текущий факториал
    F *= factorial;
}

# Выводим результат
std::cout << "F(" << n << ") = " << F << std::endl;

# Завершаем программу успешно
return 0;

#include <iostream>

int main() {

#константа, чтобы массив остался неизменным
    const int A[] = {1, 3, 5, 7, 9};
    const int B[] = {2, 4, 6, 8, 10, 12};

#сравниваем два массива по байтам
    const int a = sizeof(A) / sizeof(A[0]);
    const int b = sizeof(B) / sizeof(B[0]);

    int i = 0; #указатель на текущий элемент в A
    int j = 0; #указатель на текущий элемент в B

    #проходит по массиву столько раз, сколько всего элементов в обоих массивах
    for (int count = 0; count < a + b; ++count) {
        #Если оба массива ещё не кончились — сравниваем и берём меньший
        if (i < a && j < b) { #защищает от выхода за границы массива, когда один из них уже закончился.
            if (A[i] <= B[j]) { #элементы из А идут раньше чем равные им из В
                std::cout << A[i] << " ";
                ++i;
            } else {
                std::cout << B[j] << " ";
                ++j;
            }
        }
        # Если B уже закончился — просто выводим остаток из A
        else if (i < a) {
            std::cout << A[i] << " ";
            ++i;
        }
        #А если A закончился — добираем из B
        else {
            std::cout << B[j] << " ";
            ++j;
        }
    }

    std::cout << std::endl;
    return 0;
}
